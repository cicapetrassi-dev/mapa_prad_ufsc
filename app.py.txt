# 2. Carregamento do Shapefile Real do Campus
@st.cache_data
def carregar_dados_prad():
    caminho_shapefile = "Climácicas.shp"
    
    try:
        # Lê o shapefile real usando GeoPandas
        gdf = gpd.read_file(caminho_shapefile)
        
        # Garante o sistema de coordenadas padrão web (WGS84)
        if gdf.crs != "EPSG:4326":
            gdf = gdf.to_crs("EPSG:4326")
            
        # Mapeia as colunas reais do seu arquivo para o painel
        renomear = {}
        if "Nome" in gdf.columns: renomear["Nome"] = "id"
        if "Classifica" in gdf.columns: renomear["metodo"] = "Classifica"
        if "Área_comp" in gdf.columns: renomear["Área_comp"] = "area_ha"
        
        gdf = gdf.rename(columns=renomear)
        
        # Cria colunas de suporte caso não existam no shapefile
        if "id" not in gdf.columns:
            gdf["id"] = [f"Talhao-{i+1}" for i in range(len(gdf))]
        if "area_ha" not in gdf.columns:
            gdf["area_ha"] = 1.0
        if "status" not in gdf.columns:
            gdf["status"] = "2. Plantio Executado"
        if "setor" not in gdf.columns:
            gdf["setor"] = "Campus Principal"
        if "especies" not in gdf.columns:
            gdf["especies"] = "Espécies Nativas da Mata Atlântica"
        if "descricao_fase" not in gdf.columns:
            gdf["descricao_fase"] = "Área oficial mapeada pelo PRAD em monitoramento."
            
        # Converte para GeoJSON e DataFrame
        geojson_data = json.loads(gdf.to_json())
        df = pd.DataFrame(gdf.drop(columns="geometry"))
        
        return df, geojson_data

    except Exception as e:
        st.error(f"Erro ao carregar o Shapefile 'Climácicas.shp': {e}")
        st.info("Certifique-se de que todos os arquivos do shapefile (.shp, .dbf, .shx, .prj) estão na mesma pasta do arquivo app.py.")
        return pd.DataFrame(), {}

# Carrega os dados reais
df_prad, geojson_data = carregar_dados_prad()

# Se houver dados carregados com sucesso
if not df_prad.empty:
    # 3. Painel de Métricas de Impacto (KPIs)
    st.markdown("### 📊 Indicadores Globais do Programa")
    col1, col2, col3, col4 = st.columns(4)
    col1.metric("Área Total em Recuperação", f"{df_prad['area_ha'].sum():.2f} hectares")
    col2.metric("Talhões Mapeados", f"{len(df_prad)} glebas")
    col3.metric("Ciclo de Ações", "2023 – Presente")
    col4.metric("Espécies Nativa Foco", "Mata Atlântica")

    st.markdown("---")

    # 4. Filtros na Barra Lateral
    st.sidebar.header("🔍 Filtros do Mapa")
    status_disponiveis = sorted(df_prad["status"].unique())
    status_selecionados = st.sidebar.multiselect(
        "Filtrar por Estágio do PRAD:",
        options=status_disponiveis,
        default=status_disponiveis
    )

    df_filtrado = df_prad[df_prad["status"].isin(status_selecionados)]

    # 5. Layout Principal: Mapa e Legenda
    col_mapa, col_info = st.columns([2, 1])

    with col_mapa:
        st.subheader("🗺️ Mapeamento Espacial das Áreas (Shapefile Real)")
        if not df_filtrado.empty and geojson_data:
            fig = px.choropleth(
                df_filtrado,
                geojson=geojson_data,
                locations='id',
                featureidkey='id',
                color='status',
                hover_name='id',
                hover_data=['setor', 'area_ha', 'metodo', 'especies'],
                color_discrete_map={
                    "1. Em Preparação": "#f39c12",
                    "2. Plantio Executado": "#f1c40f",
                    "3. Em Manutenção": "#2ecc71",
                    "4. Em Monitoramento Avançado": "#27ae60"
                }
            )

            fig.update_geos(fitbounds="locations", visible=True)
            fig.update_layout(
                margin={"r":0,"t":0,"l":0,"b":0},
                legend_title_text="Estágio do PRAD"
            )
            st.plotly_chart(fig, use_container_width=True)
        else:
            st.warning("Nenhuma área selecionada para exibição no mapa.")

    with col_info:
        st.subheader("📖 Entenda as Fases do PRAD")
        st.markdown("""
        <div class="card-fase">
            <b>1. Em Preparação (Laranja):</b> Limpeza, isolamento contra impactos e controle de espécies exóticas/invasoras.
        </div>
        <div class="card-fase">
            <b>2. Plantio Executado (Amarelo):</b> Inserção de mudas nativas com espaçamento técnico e adubação inicial.
        </div>
        <div class="card-fase">
            <b>3. Em Manutenção (Verde Claro):</b> Roçadas seletivas, coroamento das mudas e replantio de reforço.
        </div>
        <div class="card-fase">
            <b>4. Monitoramento (Verde Escuro):</b> Fechamento de dossel e regeneração natural assistida.
        </div>
        """, unsafe_allow_html=True)

    # 6. Tabela Detalhada de Atributos
    with st.expander("📋 Ver Tabela Detalhada de Atributos dos Talhões"):
        st.dataframe(
            df_filtrado.rename(columns={
                "id": "Identificação (Nome)",
                "setor": "Setor de Referência",
                "status": "Status Atual",
                "area_ha": "Área (ha)",
                "metodo": "Classificação / Método",
                "especies": "Espécies Chave",
                "descricao_fase": "Descrição Pedagógica"
            }),
            use_container_width=True
        )
else:
    st.error("Não foi possível carregar os dados do Shapefile. Verifique se os arquivos estão na mesma pasta.")
